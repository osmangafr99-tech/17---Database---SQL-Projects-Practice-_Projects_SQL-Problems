select  top (3) * from (
SELECT Makes.Make,COUNT(*) AS NumberOfModels
FROM Makes
INNER JOIN
MakeModels ON Makes.MakeID = MakeModels.MakeID
GROUP BY Makes.Make
) R1
order by R1.NumberOfModels desc
