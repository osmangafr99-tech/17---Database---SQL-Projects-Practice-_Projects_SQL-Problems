select  VehicleDetails.Vehicle_Display_Name,VehicleDetails.Year,calculatetheage = YEAR(GETDATE()) - VehicleDetails.Year
from VehicleDetails
where YEAR(GETDATE()) - VehicleDetails.Year Between 15 and 25
order by calculatetheage
