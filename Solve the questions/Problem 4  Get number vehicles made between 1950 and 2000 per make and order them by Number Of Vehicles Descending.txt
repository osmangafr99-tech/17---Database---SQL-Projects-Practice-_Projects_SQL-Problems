SELECT Makes.Make,NumberOfVehicles = Count(*) 
FROM VehicleDetails
INNER JOIN
Makes ON VehicleDetails.MakeID = Makes.MakeID
WHERE VehicleDetails.Year BETWEEN 1950 AND 2000
GROUP BY Makes.Make
ORDER BY NumberOfVehicles desc
